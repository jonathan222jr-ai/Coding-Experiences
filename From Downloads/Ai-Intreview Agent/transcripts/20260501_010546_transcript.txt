  ANSWER: Sort intervals by start time, then iterate — if current interval overlaps or is adjacent to the last merged one, extend it. Otherwise push a new interval. "Adjacent" means one interval ends exactly where the next begins (200+1 = 201).
  
  CODE:
  ```typescript
  function mergeIntervals(intervals: number[][]): number[][] {
    if (!intervals.length) return [];
    
    // Sort by start time
    intervals.sort((a, b) => a[0] - b[0]);
    
    const merged: number[][] = [intervals[0]];
    
    for (let i = 1; i < intervals.length; i++) {
      const last = merged[merged.length - 1];
      const [currStart, currEnd] = intervals[i];
      
      // Overlap OR adjacent (last[1] + 1 handles 200,201 case)
      if (currStart <= last[1] + 1) {
        last[1] = Math.max(last[1], currEnd); // extend if needed
      } else {
        merged.push([currStart, currEnd]);
      }
    }
    
    return merged;
  }
  ```
